class Solution {
public:
    int removeDuplicates(vector<int>& nums) {
        
        vector<int> aws;
        int k = 0;
        int temp = -101;

        
        for(int i = 0; i < nums.size() ; i++)
        {                
            if( temp < nums[i])
            {
                aws.push_back(nums[i]);
                k++;
            }
            temp = nums[i];
        }
        
        nums = aws;
        return k;
    }
};
