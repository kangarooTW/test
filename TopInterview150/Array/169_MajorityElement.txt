class Solution {
public:
    int majorityElement(vector<int>& nums) {
        
        int aws = 0;
        int max_num = 0;
        unordered_map <int, int> mp;
        
        for (int i = 0; i< nums.size(); i++)
        {
            if(mp.count(nums[i]))
            {
                mp[nums[i]]++;
            }
            else
            {
                mp[nums[i]] = 1;
            }
        }
        
        for(const auto& n : mp)
        {
            if (max_num < n.second)
            {   
                max_num = max( max_num, n.second);
                aws = n.first;
            }
        }       
           

        return aws;
    }
};
