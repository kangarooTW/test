//Time  :O(n)
//space :O(1)

class Solution {
public:
    int maxProfit(vector<int>& prices) {
        
        int temp_min = INT_MAX;
        int profit = 0;
        for (auto i:prices)
        {
            temp_min = min(temp_min, i);
            profit = max(i - temp_min, profit);
        }
        
        return profit;
    }
};
