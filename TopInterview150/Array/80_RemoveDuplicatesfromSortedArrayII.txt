class Solution {
public:
    int removeDuplicates(vector<int>& nums) {
        
        int k = 1;
        int count = 0;
        
        for (int i = 1; i < nums.size(); i++)
        {
        k++;                  
            if(nums[i] == nums[i-1])
            {    
                //cout << "nums[i]: " << nums[i] << "\n";
                //cout << "nums[i-1]: " << nums[i-1] << "\n";
                count++;
                if (count > 1)
                {
                    //cout << "count: " << count << "\n";
                    nums.erase(nums.begin() + i);
                    i--;
                    k--;   
                }
            }
            else
                count = 0;
        }         
        return k; 
    }
};
