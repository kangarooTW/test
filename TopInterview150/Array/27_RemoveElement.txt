class Solution {
public:
    int removeElement(vector<int>& nums, int val) {
        vector<int> aws;
        int k = 0;
        int temp =0;
        for(int i =0; i < nums.size() ; i++)
        {
            //check nums[i] == val? 
            if(nums[i] == val )
            {
                temp++;
            }
            else
            {
                aws.push_back(nums[i]);
            }        
        }
        k = nums.size() - temp;
        nums = aws;

        return k;
    }
