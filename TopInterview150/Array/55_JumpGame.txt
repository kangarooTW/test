//Time  : O(n)
//Space : O(n)


class Solution {
public:
    bool canJump(vector<int>& nums) {
        
        //DFS
        stack<int> s;
        vector<bool> travel(nums.size(), false);        
        int start = 0;
        int target = nums.size()-1;
        int pos = 0;

        s.push(start);
        while (!s.empty())
        {      
            pos = s.top();
            if(s.top() == target )
            {    
                return true;
            }
            else if (!s.empty()) 
            {
                s.pop();
            }

            for(int j = 1;(j <= nums[pos])&&((j + pos) <nums.size()); j++)
            {
                //put new node in stack
                if ((!travel[j + pos]))
                {    
                    s.push(j + pos);
                    travel[j + pos] = true;
                }
            }
        }
        return false;
    }
};
